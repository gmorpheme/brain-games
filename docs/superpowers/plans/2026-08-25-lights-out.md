# Lights Out Game Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a classic Lights Out puzzle game (`lights.html`) to the brain-games site, mobile-friendly from day one, with a menu card on `index.html`.

**Architecture:** One new self-contained HTML file following the site's established pattern (inline CSS, markup, single script block). The board is a CSS grid of `<button>` cells — no canvas — so click, tap, and keyboard all work natively. Game state is a flat boolean array; scrambles are generated by simulated presses so every puzzle is solvable.

**Tech Stack:** Vanilla HTML/CSS/JS. No build tooling, no test framework. Verification via Playwright MCP tools in a live browser.

**Spec:** `docs/superpowers/specs/2026-08-25-lights-out-design.md`

## Global Constraints

- No build tooling; `lights.html` fully self-contained (inline CSS + one script block).
- Green-on-black terminal aesthetic (`#00ff00` on `#0a0a0a`/`#000`, `Courier New`, glow shadows).
- Responsive breakpoint `@media (max-width: 800px)`, matching the other pages.
- Board size selectable 3×3–7×7 (slider, default 5), best score (fewest moves) per size.
- `localStorage` keys: `lightsOutSettings` (`{ gridSize }`), `lightsOutBestScores` (`{ [gridSize]: fewestMoves }`).
- Scramble: K = `Math.ceil(gridSize * gridSize / 2)` presses on K distinct cells; par = K; re-scramble if the result is all-off.
- Depends on the mobile-support work being merged first (both touch `index.html`).

## Verification Setup (used by every task)

Serve the checkout with `python3 -m http.server 8765 --directory <checkout root>` (leave running; skip if the controller says a server is already running). Load Playwright MCP tools via ToolSearch: `select:mcp__playwright__browser_navigate,mcp__playwright__browser_resize,mcp__playwright__browser_take_screenshot,mcp__playwright__browser_evaluate,mcp__playwright__browser_click,mcp__playwright__browser_snapshot`. Phone viewport = 390×844; desktop = 1280×900. The page's top-level `let` bindings (`cells`, `moves`, `gridSize`, …) and functions (`applyPress`, `render`, …) are directly accessible by name from `browser_evaluate`.

    <script>
        const board = document.getElementById('board');

        // Game state
        let gridSize = 5;
        let cells = [];
        let moves = 0;
        let par = 0;
        let gameWon = false;
        let bestScores = {};

        // UI elements
        const movesDisplay = document.getElementById('moves');
        const parDisplay = document.getElementById('par');
        const bestScoreDisplay = document.getElementById('bestScore');
        const bestScoreDetailDisplay = document.getElementById('bestScoreDetail');
        const resultMessage = document.getElementById('resultMessage');
        const newGameButton = document.getElementById('newGameButton');
        const clearScoresButton = document.getElementById('clearScoresButton');
        const gridSizeSlider = document.getElementById('gridSize');
        const gridSizeDisplay = document.getElementById('gridSizeDisplay');

        // Toggle a cell and its four orthogonal neighbours
        function applyPress(index) {
            const row = Math.floor(index / gridSize);
            const col = index % gridSize;
            const toggle = (r, c) => {
                if (r >= 0 && r < gridSize && c >= 0 && c < gridSize) {
                    cells[r * gridSize + c] = !cells[r * gridSize + c];
                }
            };
            toggle(row, col);
            toggle(row - 1, col);
            toggle(row + 1, col);
            toggle(row, col - 1);
            toggle(row, col + 1);
        }

        // Scramble by pressing K distinct random cells (always solvable)
        function scramble() {
            const total = gridSize * gridSize;
            par = Math.ceil(total / 2);
            do {
                cells = new Array(total).fill(false);
                const indices = [...Array(total).keys()];
                for (let i = 0; i < par; i++) {
                    const j = i + Math.floor(Math.random() * (indices.length - i));
                    [indices[i], indices[j]] = [indices[j], indices[i]];
                    applyPress(indices[i]);
                }
            } while (cells.every(c => !c));
        }

        // Rebuild the board's DOM for the current grid size
        function buildBoard() {
            board.innerHTML = '';
            board.style.gridTemplateColumns = `repeat(${gridSize}, 1fr)`;
            for (let i = 0; i < gridSize * gridSize; i++) {
                const cell = document.createElement('button');
                cell.className = 'cell';
                cell.addEventListener('click', () => cellPressed(i));
                board.appendChild(cell);
            }
        }

        // Sync cell elements with state
        function render() {
            for (let i = 0; i < cells.length; i++) {
                board.children[i].classList.toggle('lit', cells[i]);
            }
        }

        function cellPressed(index) {
            if (gameWon) return;
            applyPress(index);
            moves++;
            render();
            updateDisplay();
            checkWin();
        }

        function checkWin() {
            if (cells.every(c => !c)) {
                gameWon = true;
                const isNewBest = saveBestScore();
                resultMessage.textContent = isNewBest ? '✓ NEW BEST!' : '✓ SOLVED!';
                resultMessage.className = 'success';
                updateDisplay();
            }
        }

        function initGame() {
            scramble();
            moves = 0;
            gameWon = false;
            resultMessage.textContent = '';
            resultMessage.className = '';
            buildBoard();
            render();
            updateDisplay();
        }

        function loadSettings() {
            const saved = localStorage.getItem('lightsOutSettings');
            if (saved) {
                const settings = JSON.parse(saved);
                gridSize = settings.gridSize || 5;
                gridSizeSlider.value = gridSize;
                gridSizeDisplay.textContent = gridSize;
            }

            const savedScores = localStorage.getItem('lightsOutBestScores');
            if (savedScores) {
                bestScores = JSON.parse(savedScores);
            }
        }

        function saveSettings() {
            const settings = { gridSize };
            localStorage.setItem('lightsOutSettings', JSON.stringify(settings));
        }

        function saveBestScore() {
            const currentBest = bestScores[gridSize];
            if (!currentBest || moves < currentBest) {
                bestScores[gridSize] = moves;
                localStorage.setItem('lightsOutBestScores', JSON.stringify(bestScores));
                return true;
            }
            return false;
        }

        function updateDisplay() {
            movesDisplay.textContent = moves;
            parDisplay.textContent = par;

            const bestScore = bestScores[gridSize];
            if (bestScore !== undefined) {
                bestScoreDisplay.textContent = bestScore;
                const bestDiff = bestScore - par;
                bestScoreDetailDisplay.textContent = bestDiff <= 0 ? 'AT OR UNDER PAR!' : `(par +${bestDiff})`;
            } else {
                bestScoreDisplay.textContent = '--';
                bestScoreDetailDisplay.textContent = '';
            }
        }

        // Button handlers
        newGameButton.addEventListener('click', () => {
            initGame();
        });

        clearScoresButton.addEventListener('click', () => {
            if (confirm('Clear all best scores?')) {
                bestScores = {};
                localStorage.removeItem('lightsOutBestScores');
                updateDisplay();
            }
        });

        gridSizeSlider.addEventListener('input', (e) => {
            gridSize = parseInt(e.target.value);
            gridSizeDisplay.textContent = gridSize;
            saveSettings();
            initGame();
        });

        // Initialize
        loadSettings();
        initGame();
    </script>
</body>
</html>
